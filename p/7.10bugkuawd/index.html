def check_url(url):
    try:
        response = requests.get(url, timeout=5)
        if response.status_code == 200:
            print(f"Valid URL: {url}")  # 只打印有效的URL
            return True
    except requests.exceptions.RequestException:
        pass  # 不打印任何错误信息
    return False
